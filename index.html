Step 2 — CORS Proxy (free Cloudflare Worker):
1️⃣ Go to cloudflare.com → Sign up free → Workers & Pages → Create
2️⃣ Paste this worker code → Deploy:
addEventListener('fetch',e=>e.respondWith(handle(e.request)));
async function handle(req){
 if(req.method==='OPTIONS')return new Response(null,{headers:{'Access-Control-Allow-Origin':'*','Access-Control-Allow-Methods':'POST','Access-Control-Allow-Headers':'Content-Type'}});
 const{token,userId,msg}=await req.json();
 const r=await fetch('https://api.line.me/v2/bot/message/push',{method:'POST',headers:{'Authorization':'Bearer '+token,'Content-Type':'application/json'},body:JSON.stringify({to:userId,messages:[{type:'text',text:msg}]})});
 return new Response(JSON.stringify({ok:r.ok}),{headers:{'Content-Type':'application/json','Access-Control-Allow-Origin':'*'}});
}
